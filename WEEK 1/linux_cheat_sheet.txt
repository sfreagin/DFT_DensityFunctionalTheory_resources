#!/bin/bash								#required line
#SBATCH --job-name=hello-				#set the job name to "hello"
#SBATCH --output=hello-%j.out 			#the output will be written to hello-71515.out (the JOB_ID e.g. "71515" will replace %j)
#SBATCH --error=hello-%j.err 			#any runtime errors will create a hello-71515.err file
#SBATCH --nodes=1 						#request 1 physical node (a node is the physical server)
#SBATCH --ntasks=4						#request 4 tasks total, these are the MPI processes that will launch
										# 	because --nodes=1, all 4 tasks run on that single node
#SBATCH --cpus-per-task=1 				#each task gets 1 core (1 CPU) for its own threads
#SBATCH --time=0-00:05:00 				#time limit for the job, in this case 5 minutes
										#	days-hours:minutes:seconds
#SBATCH --mem-per-cpu=1000				#request 1000MB (~1GB) of memory per CPU, or 4GB total here
#SBATCH --partition=cpuqs 				#submits job to the `cpuqs` partition rather than default
#SBATCH --exclusive 					#request exclusive access to allocated node
										#	no other jobs will be scheduled, even with leftover capacity

echo "Job ID: $SLURM_JOB_ID"			#print the JOB_ID
echo "Number of tasks: $SLURM_NTASKS"	#print the number of tasks

mpirun bash -c 'echo "Hello world from process $SLURM_PROCID of $SLURM_NTASKS on host $(hostname)"'
										#this launches the actual parallel work via MPI
										#	mpirun starts one copy of the given command per task, so 4 copies here
										#	bash -c '...' runs small inline bash command in each copy
										#	inside of that command:
										#		* $SLURM_PROCID is particular task rank/index (0 through 3)
										#		* $SLURM_NTASKS is 4 again (because --ntasks=4)
										#		* $(hostname) print node it's actually running on

#### 1. copy/paste this from the si.scf.out output file:
grep -e 'total energy' -e estimated si.scf.out | \
awk '/total energy/{e=$NF=="Ry"?$(NF-1):$NF} /estimated/{print NR, e, $(NF-1)}' \
> scf_convergence.dat


#### 2. save this as plot_scf.gp
set terminal pngcairo size 900,600
set output 'scf_convergence.png'

set multiplot layout 1,2 title "SCF Convergence"

set xlabel "Iteration"
set ylabel "Total Energy (Ry)"
set title "Total Energy"
plot 'scf_convergence.dat' using 1:2 with linespoints pt 7 lc rgb 'blue' notitle

set logscale y
set ylabel "Estimated SCF Accuracy (Ry)"
set title "Convergence (log scale)"
plot 'scf_convergence.dat' using 1:3 with linespoints pt 7 lc rgb 'red' notitle

unset multiplot
####

#### 3. run this plotting command to create scf_convergence.png
gnuplot plot_scf.gp
